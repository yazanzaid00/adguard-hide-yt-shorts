! YT Homepage and Subscriptions – Hide the Shorts section
youtube.com##ytd-rich-section-renderer[is-shorts]
youtube.com##ytd-rich-section-renderer:has(#rich-shelf-header:has-text(Shorts))
youtube.com##[is-shorts]
! YT Menu – Hide the Shorts button
youtube.com###guide [title="Shorts"], .ytd-mini-guide-entry-renderer[title="Shorts"]
youtube.com###guide [title="Shorts"]
! YT Search – Hide Shorts (video results with Shorts links)
youtube.com##ytd-video-renderer:has(a[href*="/shorts/"])
! YT Search, Channels, Sidebar – Hide Shorts sections
youtube.com##ytd-reel-shelf-renderer
! YT Channels – Hide the Shorts tab (old layout)
youtube.com##tp-yt-paper-tab[aria-label="Shorts"]
! YT Channels – Hide the Shorts tab (new layout 2023+)
youtube.com##yt-tab-shape:has-text(/^Shorts$/)
! YT Subscriptions – Hide Shorts (Grid view)
youtube.com##ytd-grid-video-renderer:has(a[href*="/shorts/"])
! YT Subscriptions – Hide Shorts (List view)
youtube.com##ytd-video-renderer:has(a[href*="/shorts/"])
! YT Subscriptions – Hide Shorts (New layout)
youtube.com##ytd-rich-item-renderer:has(a[href*="/shorts/"])
! YT Sidebar – Hide Shorts suggestions
youtube.com##ytd-compact-video-renderer:has(a[href*="/shorts/"])
! YT Collapsed Sidebar (additional rule for dynamic cases)
youtube.com##ytd-mini-guide-entry-renderer:has(a[title="Shorts"])

m.youtube.com##ytm-compact-video-renderer:has([data-style="SHORTS"])

! YT Homepage – Hide empty Shorts section (after other rules applied)
youtube.com##ytd-rich-section-renderer.style-scope.ytd-rich-grid-renderer
! YT Desktop – Hide Shorts link in mini guide
youtube.com##a#endpoint.yt-simple-endpoint.style-scope.ytd-mini-guide-entry-renderer[title="Shorts"]

! YT Desktop – Hide Shorts link in main guide
youtube.com##a#endpoint.yt-simple-endpoint.style-scope.ytd-guide-entry-renderer[title="Shorts"]
